cmake_minimum_required(VERSION 3.22)
project(TakaoQt VERSION 1.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8.3 REQUIRED COMPONENTS Widgets Network Svg)
qt_add_executable(Takao.Qt WIN32 main.cpp ApiClient.cpp DataModel.cpp Chart.cpp MainWindow.cpp Actions.cpp Settings.cpp Takao.rc)
target_link_libraries(Takao.Qt PRIVATE Qt6::Widgets Qt6::Network Qt6::Svg)
if(MSVC)
  target_compile_options(Takao.Qt PRIVATE /utf-8 /W4 /permissive-)
endif()
