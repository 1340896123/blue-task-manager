cmake_minimum_required(VERSION 3.22)
project(TakaoQt VERSION 1.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8.3 REQUIRED COMPONENTS Widgets Network Svg)
add_library(Takao.Qt.Ui STATIC ApiClient.cpp IconCache.cpp DataModel.cpp Chart.cpp MainWindow.cpp Actions.cpp Settings.cpp FluentStyle.cpp FluentDialogs.cpp)
target_include_directories(Takao.Qt.Ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(Takao.Qt.Ui PUBLIC Qt6::Widgets Qt6::Network Qt6::Svg)
qt_add_executable(Takao.Qt WIN32 main.cpp Takao.rc)
target_link_libraries(Takao.Qt PRIVATE Takao.Qt.Ui)
option(TAKAO_BUILD_INTEGRATION_TESTS "Build real backend/native component integration harness" OFF)
if(TAKAO_BUILD_INTEGRATION_TESTS)
  qt_add_executable(Takao.Qt.PageCapture WIN32 ../../tests/qt-page-capture.cpp Takao.rc)
  target_link_libraries(Takao.Qt.PageCapture PRIVATE Takao.Qt.Ui)
  qt_add_executable(Takao.Qt.Integration WIN32 ../../tests/qt-integration.cpp Takao.rc)
  target_link_libraries(Takao.Qt.Integration PRIVATE Takao.Qt.Ui)
  qt_add_executable(Takao.Qt.Fluent.Tests WIN32 ../../tests/qt-fluent-controls.cpp Takao.rc)
  target_link_libraries(Takao.Qt.Fluent.Tests PRIVATE Takao.Qt.Ui)
  qt_add_executable(Takao.Qt.IconCache.Tests ../../tests/qt-icon-cache.cpp)
  target_link_libraries(Takao.Qt.IconCache.Tests PRIVATE Takao.Qt.Ui)
endif()
if(MSVC)
  target_compile_options(Takao.Qt.Ui PUBLIC /utf-8 /W4 /permissive-)
endif()
